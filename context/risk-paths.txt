# Paths that make a changed file HIGH risk for scripts/diff-scope.sh.
# One extended regex per line, matched against the LOWERCASED file path. Blank lines and
# lines starting with # are ignored. HIGH-risk files are read first, are never dropped by
# a --budget cut-off, and are the ones that get the full mandatory-check treatment.
#
# Keep this list short and specific to your codebase: every pattern that matches too much
# turns "high risk" into "everything" and the saving disappears. Edit freely. The defaults
# below are generic; add your stack's own entries (examples in the comments).

# auth / session / credentials
(^|/)[^/]*(auth|login|logout|session|token|keychain|credential|oauth|biometric|secret)
# persistence / schema
(^|/)[^/]*(persistence|migration|database|schema|repository)
# permission / capability config, reviewed like source code
(^|/)[^/]*(permission|entitlement|capabilit)
# concurrency primitives
(^|/)[^/]*(actor|dispatch|concurren|mutex|semaphore|thread|lock)
# deployment / pipeline config
(^|/)(dockerfile|\.github/workflows/|azure-pipelines)
# stack examples (uncomment what applies):
# \.xcdatamodel                 # Core Data models
# (^|/)info\.plist$             # iOS app config
# (^|/)androidmanifest\.xml$    # Android app config
# (^|/)(build\.gradle|package\.json|pom\.xml)$   # dependency manifests
